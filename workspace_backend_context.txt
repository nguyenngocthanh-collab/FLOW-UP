import type { IncomingMessage, ServerResponse } from "node:http";

export function sendJson(
  res: ServerResponse,
  statusCode: number,
  data: unknown,
  corsOrigin = "*"
): void {
  if (res.writableEnded) return;

  const jsonString = JSON.stringify(data);
  res.writeHead(statusCode, {
    "Content-Type": "application/json; charset=utf-8",
    "Content-Length": Buffer.byteLength(jsonString),
    "Access-Control-Allow-Origin": corsOrigin,
    "Access-Control-Allow-Methods": "GET, POST, PATCH, PUT, DELETE, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Workspace-Id",
    "Access-Control-Max-Age": "86400"
  });
  res.end(jsonString);
}

export function sendSuccess<T>(
  res: ServerResponse,
  data: T,
  statusCode = 200,
  corsOrigin = "*"
): void {
  sendJson(res, statusCode, { success: true, data }, corsOrigin);
}

export function sendError(
  res: ServerResponse,
  statusCode: number,
  code: string,
  message: string,
  corsOrigin = "*"
): void {
  sendJson(res, statusCode, { success: false, error: { code, message } }, corsOrigin);
}

export function getStatusCodeForError(code: string): number {
  if (code.includes("RATE_LIMITED")) return 429;
  if (
    code.includes("INVALID_CREDENTIALS") ||
    code.includes("EMAIL_NOT_CONFIRMED") ||
    code.includes("UNAUTHENTICATED")
  ) return 401;
  if (code.includes("PROVIDER_ERROR") || code.includes("NETWORK_ERROR")) return 502;
  if (code.includes("NOT_FOUND")) return 404;
  if (code.includes("ACCESS_DENIED") || code.includes("OWNERSHIP")) return 403;
  if (code.includes("ALREADY_EXISTS") || code.includes("CIRCULAR") || code.includes("CONFLICT")) return 409;
  if (code.includes("INVALID") || code.includes("INPUT") || code.includes("REQUIRED")) return 400;
  return 400;
}

export async function parseJsonBody(req: IncomingMessage): Promise<unknown> {
  return new Promise((resolve, reject) => {
    let raw = "";
    req.setEncoding("utf-8");

    req.on("data", (chunk: string) => {
      raw += chunk;
      if (raw.length > 1024 * 1024) {
        // 1MB limit
        reject(new Error("Request payload too large"));
      }
    });

    req.on("end", () => {
      if (!raw.trim()) {
        resolve({});
        return;
      }
      try {
        const parsed = JSON.parse(raw);
        resolve(parsed);
      } catch {
        reject(new Error("Invalid JSON body"));
      }
    });

    req.on("error", (err) => {
      reject(err);
    });
  });
}

export function extractBearerToken(req: IncomingMessage): string | null {
  const authHeader = req.headers.authorization;
  if (!authHeader) return null;

  const parts = authHeader.trim().split(/\s+/);
  if (parts.length === 2 && parts[0].toLowerCase() === "bearer") {
    return parts[1];
  }
  return null;
}

export function createApiHandler(
  container: ApplicationContainer,
  corsOrigin = "*"
): (req: IncomingMessage, res: ServerResponse) => Promise<void> {
  const router = createRouter();

  return async (req: IncomingMessage, res: ServerResponse): Promise<void> => {
    // 1. Handle CORS preflight
    if (req.method === "OPTIONS") {
      res.writeHead(204, {
        "Access-Control-Allow-Origin": corsOrigin,
        "Access-Control-Allow-Methods": "GET, POST, PATCH, PUT, DELETE, OPTIONS",
        "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Workspace-Id",
        "Access-Control-Max-Age": "86400"
      });
      res.end();
      return;
    }

    try {
      const parsedUrl = new URL(req.url ?? "/", "http://localhost");
      const pathname = parsedUrl.pathname;
      const query = parsedUrl.searchParams;
      const method = req.method ?? "GET";

      const match = router.match(method, pathname);
      if (!match) {
        sendError(
          res,
          404,
          "ROUTE_NOT_FOUND",
          `Route not found: ${method} ${pathname}`,
          corsOrigin
        );
        return;
      }

      // 2. Authentication check
      let user: import("../modules/auth/auth.types.ts").AuthenticatedUserContext | null = null;
      const token = extractBearerToken(req);
      if (token) {
        user = await container.tokenVerifier.verifyToken(token);
        const selectedWorkspaceId = req.headers["x-workspace-id"];
        if (user && typeof selectedWorkspaceId === "string" && selectedWorkspaceId.trim()) {
          user = { ...user, workspaceId: selectedWorkspaceId.trim() };
        }
      }

      if (match.requiresAuth && !user) {
        sendError(
          res,
          401,
          "UNAUTHORIZED",
          "Authentication required. Please provide a valid Bearer token.",
          corsOrigin
        );
        return;
      }

      // 3. Body parsing for POST / PATCH / PUT
      let body: unknown = null;
      if (["POST", "PATCH", "PUT"].includes(method)) {
        try {
          body = await parseJsonBody(req);
        } catch (err: unknown) {
          sendError(
            res,
            400,
            "INVALID_JSON_BODY",
            (err as Error)?.message || "Malformed JSON body.",
            corsOrigin
          );
          return;
        }
      }

      // 4. Request context
      const ctx: RequestContext = {
        req,
        res,
        container,
        user,
        params: match.params,
        query,
        body,
        corsOrigin
      };

      // 5. Execute handler
      await match.handler(ctx);
    } catch (err: unknown) {
      console.error("[API Error]", err);
      if (!res.writableEnded) {
        sendError(
          res,
          500,
          "INTERNAL_SERVER_ERROR",
          "An unexpected server error occurred.",
          corsOrigin
        );
      }
    }
  };
}

import type { HttpMethod, RouteHandler, RouteMatch } from "./types.ts";

interface RouteRegistration {
  method: HttpMethod;
  pattern: RegExp;
  paramNames: string[];
  handler: RouteHandler;
  requiresAuth: boolean;
}

export class Router {
  private readonly routes: RouteRegistration[] = [];

  register(
    method: HttpMethod,
    pathPattern: string,
    handler: RouteHandler,
    options: { requiresAuth?: boolean } = {}
  ): void {
    const paramNames: string[] = [];
    const regexPattern = pathPattern
      .replace(/\/+$/, "") // remove trailing slash
      .replace(/:([a-zA-Z0-9_]+)/g, (_, paramName) => {
        paramNames.push(paramName);
        return "([^/]+)";
      });

    const pattern = new RegExp(`^${regexPattern || "/"}$`);
    this.routes.push({
      method,
      pattern,
      paramNames,
      handler,
      requiresAuth: options.requiresAuth ?? true
    });
  }

  get(path: string, handler: RouteHandler, options?: { requiresAuth?: boolean }): void {
    this.register("GET", path, handler, options);
  }

  post(path: string, handler: RouteHandler, options?: { requiresAuth?: boolean }): void {
    this.register("POST", path, handler, options);
  }

  patch(path: string, handler: RouteHandler, options?: { requiresAuth?: boolean }): void {
    this.register("PATCH", path, handler, options);
  }

  delete(path: string, handler: RouteHandler, options?: { requiresAuth?: boolean }): void {
    this.register("DELETE", path, handler, options);
  }

  match(method: string, pathname: string): RouteMatch | null {
    const cleanPath = pathname.length > 1 && pathname.endsWith("/") ? pathname.slice(0, -1) : pathname;

    for (const route of this.routes) {
      if (route.method !== method) continue;

      const match = route.pattern.exec(cleanPath);
      if (match) {
        const params: Record<string, string> = {};
        for (let i = 0; i < route.paramNames.length; i++) {
          params[route.paramNames[i]] = decodeURIComponent(match[i + 1]);
        }
        return {
          handler: route.handler,
          params,
          requiresAuth: route.requiresAuth
        };
      }
    }

    return null;
  }
}

export type HttpMethod = "GET" | "POST" | "PUT" | "PATCH" | "DELETE" | "OPTIONS" | "HEAD";

import type { Router } from "../router.ts";
import { sendError, sendSuccess } from "../http.ts";
import type { RouteHandler } from "../types.ts";
import type { AIOperationType, AIProposalSelection } from "../../modules/ai-orchestration/index.ts";
import { AIOrchestrationError } from "../../modules/ai-orchestration/index.ts";

function aiStatus(code: string): number {
  if (code === "AI_UNAUTHENTICATED") return 401;
  if (code === "AI_ACCESS_DENIED" || code === "AI_PROPOSAL_ACCESS_DENIED") return 403;
  if (code === "AI_PROPOSAL_NOT_FOUND") return 404;
  if (code === "AI_USAGE_LIMIT_EXCEEDED" || code === "AI_PROVIDER_RATE_LIMITED") return 429;
  if (code === "AI_PROVIDER_UNAVAILABLE") return 503;
  if (code.startsWith("AI_INVALID") || code === "AI_APPROVAL_REQUIRED" || code === "AI_OUTPUT_INVALID") return 400;
  if (code.startsWith("AI_PROVIDER") || code === "AI_REQUEST_TIMEOUT") return 502;
  return 409;
}

function bodyObject(value: unknown): Record<string, unknown> | null {
  return value !== null && typeof value === "object" && !Array.isArray(value)
    ? value as Record<string, unknown> : null;
}

export function registerAIRoutes(router: Router): void {
  const run = (operation: AIOperationType): RouteHandler => async (ctx) => {
    const body = bodyObject(ctx.body);
    if (typeof body?.input !== "string" || !body.input.trim()) {
      sendError(ctx.res, 400, "AI_INVALID_REQUEST", "A goal description is required.", ctx.corsOrigin);
      return;
    }
    try {
      const result = await ctx.container.aiService.execute(ctx.user!, operation, body.input);
      sendSuccess(ctx.res, result, 200, ctx.corsOrigin);
    } catch (error) {
      const failure = error instanceof AIOrchestrationError
        ? error : new AIOrchestrationError("AI_PROVIDER_ERROR", "AI assistance is temporarily unavailable.", error);
      sendError(ctx.res, aiStatus(failure.code), failure.code, failure.message, ctx.corsOrigin);
    }
  };

  router.post("/api/ai/goal-understanding", run("GOAL_CLARIFICATION"));
  router.post("/api/ai/feasibility", run("GOAL_FEASIBILITY"));
  router.post("/api/ai/goal-proposals", run("GOAL_TREE_PROPOSAL"));

  router.get("/api/ai/proposals/:id", async (ctx) => {
    try {
      sendSuccess(ctx.res, await ctx.container.aiService.getProposal(ctx.user!, ctx.params.id), 200, ctx.corsOrigin);
    } catch (error) {
      const failure = error instanceof AIOrchestrationError ? error : new AIOrchestrationError("AI_PERSISTENCE_ERROR", "Unable to load proposal.");
      sendError(ctx.res, aiStatus(failure.code), failure.code, failure.message, ctx.corsOrigin);
    }
  });

  router.patch("/api/ai/proposals/:id", async (ctx) => {
    const body = bodyObject(ctx.body);
    if (!body || !("proposalData" in body)) {
      sendError(ctx.res, 400, "AI_INVALID_REQUEST", "proposalData is required.", ctx.corsOrigin);
      return;
    }
    try {
      sendSuccess(ctx.res, await ctx.container.aiService.updateProposal(ctx.user!, ctx.params.id, body.proposalData), 200, ctx.corsOrigin);
    } catch (error) {
      const failure = error instanceof AIOrchestrationError ? error : new AIOrchestrationError("AI_PERSISTENCE_ERROR", "Unable to update proposal.");
      sendError(ctx.res, aiStatus(failure.code), failure.code, failure.message, ctx.corsOrigin);
    }
  });

  router.post("/api/ai/proposals/:id/reject", async (ctx) => {
    try {
      sendSuccess(ctx.res, await ctx.container.aiService.rejectProposal(ctx.user!, ctx.params.id), 200, ctx.corsOrigin);
    } catch (error) {
      const failure = error instanceof AIOrchestrationError ? error : new AIOrchestrationError("AI_PERSISTENCE_ERROR", "Unable to reject proposal.");
      sendError(ctx.res, aiStatus(failure.code), failure.code, failure.message, ctx.corsOrigin);
    }
  });

  router.post("/api/ai/proposals/:id/apply", async (ctx) => {
    const body = bodyObject(ctx.body) as { selection?: AIProposalSelection } | null;
    if (!body?.selection || !Array.isArray(body.selection.selectedItemIds)) {
      sendError(ctx.res, 400, "AI_APPROVAL_REQUIRED", "Select proposal items before approving.", ctx.corsOrigin);
      return;
    }
    try {
      sendSuccess(ctx.res, await ctx.container.aiService.approveAndApplyProposal(ctx.user!, ctx.params.id, body.selection), 200, ctx.corsOrigin);
    } catch (error) {
      const failure = error instanceof AIOrchestrationError ? error : new AIOrchestrationError("AI_PROPOSAL_APPLICATION_FAILED", "Unable to apply proposal.");
      sendError(ctx.res, aiStatus(failure.code), failure.code, failure.message, ctx.corsOrigin);
    }
  });
}

import type { Router } from "../router.ts";
import { getStatusCodeForError, sendError, sendSuccess } from "../http.ts";

export function registerAuthRoutes(router: Router): void {
  router.post(
    "/api/auth/register",
    async (ctx) => {
      const body = (ctx.body ?? {}) as {
        email?: string;
        password?: string;
      };

      if (
        typeof body.email !== "string" ||
        typeof body.password !== "string" ||
        !body.email.trim() ||
        !body.password.trim()
      ) {
        sendError(
          ctx.res,
          400,
          "AUTH_INVALID_INPUT",
          "Email and password are required.",
          ctx.corsOrigin
        );
        return;
      }

      const result = await ctx.container.authApiService.register({
        email: body.email,
        password: body.password
      });

      if (!result.success) {
        const status = getStatusCodeForError(result.error.code);

        sendError(
          ctx.res,
          status,
          result.error.code,
          result.error.message,
          ctx.corsOrigin
        );
        return;
      }

      sendSuccess(
        ctx.res,
        {
          ...result.data.registration,
          token: result.data.accessToken
        },
        201,
        ctx.corsOrigin
      );
    },
    { requiresAuth: false }
  );

  router.post(
    "/api/auth/login",
    async (ctx) => {
      const body = (ctx.body ?? {}) as {
        email?: string;
        password?: string;
      };

      if (
        typeof body.email !== "string" ||
        typeof body.password !== "string" ||
        !body.email.trim() ||
        !body.password.trim()
      ) {
        sendError(
          ctx.res,
          400,
          "AUTH_INVALID_INPUT",
          "Email and password are required.",
          ctx.corsOrigin
        );
        return;
      }

      const result = await ctx.container.authApiService.signIn({
        email: body.email,
        password: body.password
      });

      if (!result.success) {
        const status = getStatusCodeForError(result.error.code);

        sendError(
          ctx.res,
          status,
          result.error.code,
          result.error.message,
          ctx.corsOrigin
        );
        return;
      }

      if (!result.data.accessToken) {
        sendError(
          ctx.res,
          500,
          "AUTH_PROVIDER_ERROR",
          "Authentication provider did not return an access token.",
          ctx.corsOrigin
        );
        return;
      }

      sendSuccess(
        ctx.res,
        {
          user: result.data.user,
          token: result.data.accessToken
        },
        200,
        ctx.corsOrigin
      );
    },
    { requiresAuth: false }
  );

  router.post(
    "/api/auth/logout",
    async (ctx) => {
      const result = await ctx.container.authService.signOut();

      if (!result.success) {
        const status = getStatusCodeForError(result.error.code);

        sendError(
          ctx.res,
          status,
          result.error.code,
          result.error.message,
          ctx.corsOrigin
        );
        return;
      }

      sendSuccess(
        ctx.res,
        {
          message: "Signed out successfully."
        },
        200,
        ctx.corsOrigin
      );
    },
    { requiresAuth: false }
  );

  router.get(
    "/api/auth/session",
    async (ctx) => {

      "Failed to initialize Supabase Auth provider."
    );
  }

import {
  randomUUID,
  randomBytes,
  scrypt,
  timingSafeEqual
} from "node:crypto";

import type {
  AuthProvider,
  ProviderAuthEvent,
  ProviderResult,
  ProviderSession,
  ProviderUser
} from "../../modules/auth/auth.provider.ts";

interface StoredDevUser {
  id: string;
  email: string;
  passwordHash: string;
}

const PASSWORD_KEY_LENGTH = 64;

function derivePasswordHash(
  password: string,
  salt: string
): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    scrypt(password, salt, PASSWORD_KEY_LENGTH, (error, derivedKey) => {
      if (error) {
        reject(error);
        return;
      }

      resolve(derivedKey as Buffer);
    });
  });
}

async function hashPassword(password: string): Promise<string> {
  const salt = randomBytes(16).toString("hex");
  const hash = await derivePasswordHash(password, salt);

  return `${salt}:${hash.toString("hex")}`;
}

async function verifyPassword(
  password: string,
  storedHash: string
): Promise<boolean> {
  const [salt, originalHash] = storedHash.split(":");

  if (!salt || !originalHash || !/^[a-f0-9]+$/i.test(originalHash)) {
    return false;
  }

  const originalBuffer = Buffer.from(originalHash, "hex");

  if (originalBuffer.length !== PASSWORD_KEY_LENGTH) {
    return false;
  }

  const derivedHash = await derivePasswordHash(password, salt);

  return timingSafeEqual(originalBuffer, derivedHash);
}

export class LocalDevAuthProvider implements AuthProvider {
  private readonly users = new Map<string, StoredDevUser>();

  private currentSession: ProviderSession | null = null;

  private readonly listeners = new Set<
    (
      event: ProviderAuthEvent,
      session: ProviderSession | null
    ) => void
  >();

  async register(
    email: string,
    password: string
  ): Promise<
    ProviderResult<{
      user: ProviderUser | null;
      session: ProviderSession | null;
    }>
  > {
    const normalizedEmail = email.toLowerCase().trim();

    for (const user of this.users.values()) {
      if (user.email === normalizedEmail) {
        return {
          data: { user: null, session: null },
          error: new Error("User already exists")
        };
      }
    }

    const id = `dev-user-${randomUUID()}`;
    const passwordHash = await hashPassword(password);

    const storedUser: StoredDevUser = {
      id,
      email: normalizedEmail,
      passwordHash
    };

    this.users.set(id, storedUser);

    const providerUser: ProviderUser = {
      id,
      email: normalizedEmail
    };

    const session: ProviderSession = {
      user: providerUser,
      accessToken: `dev-token:${id}:${normalizedEmail}`
    };

    this.currentSession = session;
    this.notify("SIGNED_IN", session);

    return {
      data: { user: providerUser, session },
      error: null
    };
  }

  async signIn(
    email: string,
    password: string
  ): Promise<
    ProviderResult<{
      user: ProviderUser | null;
      session: ProviderSession | null;
    }>
  > {
    const normalizedEmail = email.toLowerCase().trim();

    let found: StoredDevUser | null = null;

    for (const user of this.users.values()) {
      if (user.email === normalizedEmail) {
        found = user;
        break;
      }
    }

    if (!found || !(await verifyPassword(password, found.passwordHash))) {
      return {
        data: { user: null, session: null },
        error: new Error("Invalid credentials")
      };
    }

    const providerUser: ProviderUser = {
      id: found.id,
      email: found.email
    };

    const session: ProviderSession = {
      user: providerUser,
      accessToken: `dev-token:${found.id}:${found.email}`
    };

    this.currentSession = session;
    this.notify("SIGNED_IN", session);

    return {
      data: { user: providerUser, session },
      error: null
    };
  }

  async signOut(): Promise<ProviderResult<null>> {
    this.currentSession = null;
    this.notify("SIGNED_OUT", null);

    return {
      data: null,
      error: null
    };
  }

  async getSession(): Promise<
    ProviderResult<{ session: ProviderSession | null }>
  > {
    return {
      data: { session: this.currentSession },
      error: null
    };
  }

  onAuthStateChange(
    callback: (
      event: ProviderAuthEvent,
      session: ProviderSession | null
    ) => void
  ): () => void {
    this.listeners.add(callback);

    return () => {
      this.listeners.delete(callback);
    };
  }

  private notify(
    event: ProviderAuthEvent,
    session: ProviderSession | null
  ): void {
    for (const listener of this.listeners) {
      try {
        listener(event, session);
      } catch {
        // Ignore callback errors.
      }
    }
  }
}

export function createLocalDevAuthProvider(): AuthProvider {
  return new LocalDevAuthProvider();
}

import type { AuthenticatedUserContext } from "../../modules/auth/auth.types.ts";

export interface TokenVerifier {
  verifyToken(token: string): Promise<AuthenticatedUserContext | null>;
}

export interface TokenVerifierOptions {
  supabaseUrl?: string;
  supabaseKey?: string;
  devMode?: boolean;
}

export class SupabaseOrDevTokenVerifier implements TokenVerifier {
  private readonly options: TokenVerifierOptions;

  constructor(options: TokenVerifierOptions) {
    this.options = options;
  }

  async verifyToken(
    token: string
  ): Promise<AuthenticatedUserContext | null> {
    const cleanToken = token.trim();

    if (!cleanToken) return null;

    const isDevToken =
      cleanToken.startsWith("dev:") ||
      cleanToken.startsWith("dev-token:") ||
      cleanToken.startsWith("mock-token:");

    // Development tokens are accepted only when explicitly enabled.
    if (isDevToken) {
      if (this.options.devMode !== true) return null;

      return this.parseDevToken(cleanToken);
    }

    // Supabase access tokens must always be verified with Supabase,
    // including when the application is running in development mode.
    if (!this.options.supabaseUrl || !this.options.supabaseKey) {
      return null;
    }

    try {
      const response = await fetch(
        `${this.options.supabaseUrl}/auth/v1/user`,
        {
          method: "GET",
          headers: {
            Authorization: `Bearer ${cleanToken}`,
            apikey: this.options.supabaseKey
          }
        }
      );

      if (!response.ok) return null;

      const data = (await response.json()) as {
        id?: string;
        email?: string;
      };

      if (!data?.id) return null;

      return {
        userId: data.id,
        email: data.email ?? null,
        isAuthenticated: true
      };
    } catch {
      return null;
    }
  }

  private parseDevToken(
    token: string
  ): AuthenticatedUserContext | null {
    // Format: dev-token:<userId>:<email>
    if (token.startsWith("dev-token:")) {
      const parts = token.slice("dev-token:".length).split(":");
      const userId = parts[0]?.trim();
      const email = parts[1]?.trim();

      if (!userId) return null;

      return {
        userId,
        email: email || `${userId}@example.com`,
        isAuthenticated: true
      };
    }

    // Format: mock-token:<userId>
    if (token.startsWith("mock-token:")) {
      const userId = token.slice("mock-token:".length).trim();

      if (!userId) return null;

      return {
        userId,
        email: `${userId}@example.com`,
        isAuthenticated: true
      };
    }

    // Format: dev:<userId>
    if (token.startsWith("dev:")) {
      const userId = token.slice("dev:".length).trim();

      if (!userId) return null;

      return {
        userId,
        email: `${userId}@example.com`,
        isAuthenticated: true
      };
    }

    return null;
  }
}

export function createTokenVerifier(
  options: TokenVerifierOptions
): TokenVerifier {
  return new SupabaseOrDevTokenVerifier(options);
}

export function normalizeProviderError(error: unknown): AuthError {
  const message =
    typeof error === "object" && error !== null && "message" in error
      ? String((error as { message: unknown }).message).toLowerCase()
      : "";

  const status =
    typeof error === "object" && error !== null && "status" in error
      ? Number((error as { status: unknown }).status)
      : 0;

  if (
    message.includes("already registered") ||
    message.includes("user already exists") ||
    message.includes("already been registered")
  ) return createAuthError("AUTH_EMAIL_ALREADY_EXISTS");

  if (
    message.includes("invalid login credentials") ||
    message.includes("invalid credentials")
  ) return createAuthError("AUTH_INVALID_CREDENTIALS");

  if (
    message.includes("email not confirmed") ||
    message.includes("email_not_confirmed")
  ) return createAuthError("AUTH_EMAIL_NOT_CONFIRMED");

  if (
    message.includes("rate limit") ||
    message.includes("too many requests") ||
    status === 429
  ) return createAuthError("AUTH_RATE_LIMITED");

  if (
    message.includes("network") ||
    message.includes("fetch failed") ||
    message.includes("failed to fetch")
  ) return createAuthError("AUTH_NETWORK_ERROR");

  if (status === 401) return createAuthError("AUTH_INVALID_CREDENTIALS");

  return createAuthError("AUTH_PROVIDER_ERROR");
}

import type { AuthProvider } from "./auth.provider";
import { createAuthError, normalizeProviderError } from "./auth.errors";
import type {
  AuthenticationService,
  AuthenticationState,
  AuthenticatedUserContext,
  OperationResult,
  RegistrationResult,
  RegisterInput,
  SignInInput
} from "./auth.types";
import { validateRegisterInput, validateSignInInput } from "./auth.validation";

export interface AuthenticationApiService {
  register(
    input: RegisterInput
  ): Promise<
    OperationResult<{
      registration: RegistrationResult;
      accessToken: string | null;
    }>
  >;

  signIn(
    input: SignInInput
  ): Promise<
    OperationResult<{
      user: AuthenticatedUserContext;
      accessToken: string | null;
    }>
  >;
}

function toUserContext(user: {
  id: string;
  email: string | null;
}): AuthenticatedUserContext {
  return {
    userId: user.id,
    email: user.email,
    isAuthenticated: true
  };
}

function success<T>(data: T): OperationResult<T> {
  return { success: true, data };
}

function failure<T>(
  code: Parameters<typeof createAuthError>[0]
): OperationResult<T> {
  return {
    success: false,
    error: createAuthError(code)
  };
}

export function createAuthenticationService(
  provider: AuthProvider
): AuthenticationService {
  return {
    async register(
      input: RegisterInput
    ): Promise<OperationResult<RegistrationResult>> {
      if (!validateRegisterInput(input).valid) {
        return failure("AUTH_INVALID_INPUT");
      }

      try {
        const result = await provider.register(
          input.email.trim(),
          input.password
        );

        if (result.error) {
          return {
            success: false,
            error: normalizeProviderError(result.error)
          };
        }

        if (result.data.session && result.data.user) {
          return success({
            status: "authenticated",
            user: toUserContext(result.data.user)
          });
        }

        return success({
          status: "confirmation_required",
          email: input.email.trim()
        });
      } catch (error) {
        return {
          success: false,
          error: normalizeProviderError(error)
        };
      }
    },

    async signIn(
      input: SignInInput
    ): Promise<OperationResult<AuthenticatedUserContext>> {
      if (!validateSignInInput(input).valid) {
        return failure("AUTH_INVALID_INPUT");
      }

      try {
        const result = await provider.signIn(
          input.email.trim(),
          input.password
        );

        if (result.error) {
          return {
            success: false,
            error: normalizeProviderError(result.error)
          };
        }

        if (!result.data.session || !result.data.user) {
          return failure("AUTH_INVALID_CREDENTIALS");
        }

        return success(toUserContext(result.data.user));
      } catch (error) {
        return {
          success: false,
          error: normalizeProviderError(error)
        };
      }
    },

    async signOut(): Promise<OperationResult<void>> {
      try {
        const result = await provider.signOut();

        if (result.error) {
          return {
            success: false,
            error: normalizeProviderError(result.error)
          };
        }

        return success(undefined);
      } catch (error) {
        return {
          success: false,
          error: normalizeProviderError(error)
        };
      }
    },

    async getAuthenticationState(): Promise<AuthenticationState> {
      try {
        const result = await provider.getSession();

        if (result.error || !result.data.session) {
          return {
            status: "unauthenticated",
            user: null
          };
        }

        return {
          status: "authenticated",
          user: toUserContext(result.data.session.user)
        };
      } catch {
        return {
          status: "unauthenticated",
          user: null
        };
      }
    },

    subscribeToAuthenticationState(listener) {
      return provider.onAuthStateChange((_event, session) => {
        if (session) {
          listener({
            status: "authenticated",
            user: toUserContext(session.user)
          });
        } else {
          listener({
            status: "unauthenticated",
            user: null
          });
        }
      });
    }
  };
}

      }

      try {
        const result = await provider.signIn(
          input.email.trim(),
          input.password
        );

        if (result.error) {
          return {
            success: false,
            error: normalizeProviderError(result.error)
          };
        }

        if (!result.data.session || !result.data.user) {
          return failure("AUTH_INVALID_CREDENTIALS");
        }

        return success({
          user: toUserContext(result.data.user),
          accessToken: result.data.session.accessToken ?? null
        });
      } catch (error) {
        return {
          success: false,
          error: normalizeProviderError(error)
        };
      }
    }
  };
}

const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export function validateRegisterInput(input: RegisterInput): ValidationResult {
  const errors: ValidationResult["errors"] = {};
  const email = input.email.trim();

  if (!email || !EMAIL_PATTERN.test(email)) {
    errors.email = "Email không hợp lệ.";
  }

  if (!input.password || input.password.trim().length === 0) {
    errors.password = "Mật khẩu không được để trống.";
  }

  return { valid: Object.keys(errors).length === 0, errors };
}

  const email = input.email.trim();

  if (!email || !EMAIL_PATTERN.test(email)) {
    errors.email = "Email không hợp lệ.";
  }

  if (!input.password) {
    errors.password = "Mật khẩu không được để trống.";
  }

  return { valid: Object.keys(errors).length === 0, errors };
}

import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import type {
  AuthProvider,
  ProviderAuthEvent,
  ProviderSession,
  ProviderUser
} from "./auth.provider";

function mapUser(user: {
  id: string;
  email?: string;
}): ProviderUser {
  return {
    id: user.id,
    email: user.email ?? null
  };
}

function mapSession(
  session: {
    user: {
      id: string;
      email?: string;
    };
    access_token: string;
    refresh_token: string;
    expires_at?: number;
  } | null
): ProviderSession | null {
  if (!session) return null;

  return {
    user: mapUser(session.user),
    accessToken: session.access_token,
    refreshToken: session.refresh_token,
    expiresAt: session.expires_at
  };
}

/**
 * Server-side Supabase Auth adapter.
 *
 * Each authentication operation gets its own client.
 * No shared persisted session is kept between API requests.
 */
export function createSupabaseAuthProvider(
  supabaseUrl: string,
  publishableKey: string
): AuthProvider {
  if (!supabaseUrl || !publishableKey) {
    throw new Error("Missing Supabase public configuration.");
  }

  function createAuthClient(): SupabaseClient {
    return createClient(supabaseUrl, publishableKey, {
      auth: {
        persistSession: false,
        autoRefreshToken: false,
        detectSessionInUrl: false
      }
    });
  }

  return {
    async register(email, password) {
      const client = createAuthClient();

      const { data, error } = await client.auth.signUp({
        email,
        password
      });

      return {
        data: {
          user: data.user ? mapUser(data.user) : null,
          session: mapSession(data.session)
        },
        error
      };
    },

    async signIn(email, password) {
      const client = createAuthClient();

      const { data, error } =
        await client.auth.signInWithPassword({
          email,
          password
        });

      return {
        data: {
          user: data.user ? mapUser(data.user) : null,
          session: mapSession(data.session)
        },
        error
      };
    },

    async signOut() {
      // The backend does not keep a shared Supabase session.
      // The client must discard its locally stored access token.
      return {
        data: null,
        error: null
      };
    },

    async getSession() {
      // Authentication state is verified per request by TokenVerifier.
      return {
        data: {
          session: null
        },
        error: null
      };
    },

    onAuthStateChange(
      _callback: (
        event: ProviderAuthEvent,
        session: ProviderSession | null
      ) => void
    ) {
      // There is no persistent server-side session to subscribe to.
      return () => {};
    }
  };
}

  createApplicationContainer,
  type ApplicationContainer,
  type ContainerConfig
} from "../application/container.ts";

export interface AppOptions {
  dbPath?: string;
  connection?: DatabaseConnection;
  config?: ContainerConfig;
  corsOrigin?: string;
}

export interface AppInstance {
  server: http.Server;
  container: ApplicationContainer;
  close: () => Promise<void>;
}

export function createApp(options: AppOptions = {}): AppInstance {
  const connection = options.connection ?? createSqliteDatabase({ dbPath: options.dbPath });
  const container = createApplicationContainer(connection, options.config);
  const corsOrigin = options.corsOrigin ?? "*";

  const apiHandler = createApiHandler(container, corsOrigin);
  const server = http.createServer(apiHandler);

  return {
    server,
    container,
    async close() {
      return new Promise<void>((resolve, reject) => {
        if (!server.listening) {
          try {
            connection.close();
          } catch {
            // ignore if already closed
          }
          resolve();
          return;
        }

        server.close((err) => {
          try {
            connection.close();
          } catch {
            // ignore if already closed
          }
          if (err) reject(err);
          else resolve();
        });
      });
    }
  };
}

// Simple .env reader if .env file exists
function loadEnv(): void {
  const envPath = path.resolve(process.cwd(), ".env");

  if (fs.existsSync(envPath)) {
    try {
      const content = fs.readFileSync(envPath, "utf-8");

      for (const line of content.split("\n")) {
        const trimmed = line.trim();

        if (!trimmed || trimmed.startsWith("#")) continue;

        const eqIdx = trimmed.indexOf("=");

        if (eqIdx !== -1) {
          const key = trimmed.slice(0, eqIdx).trim();
          const val = trimmed
            .slice(eqIdx + 1)
            .trim()
            .replace(/^["']|["']$/g, "");

          if (!(key in process.env)) {
            process.env[key] = val;
          }
        }
      }
    } catch {
      // ignore
    }
  }
}

function positiveLimit(name: string, fallback: number, max: number): number {
  const value = Number(process.env[name] ?? fallback);
  return Number.isFinite(value) ? Math.max(1, Math.min(max, Math.trunc(value))) : fallback;
}
